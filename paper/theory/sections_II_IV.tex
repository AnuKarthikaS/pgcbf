\section{Problem Formulation and System Contracts}
\label{sec:problem}

Consider a nonlinear control-affine dynamical system evolving on a full-order state space \(\mathcal{X} \subseteq \mathbb{R}^N\) (\(N \gg 1\)), governed by the semi-discretized differential equation
\begin{equation}
\label{eq:full_system}
dx(t) = \Big( F(x(t)) + G\,u(t) + D\,w(t) \Big) dt + \Sigma(x(t))\,dW_t,
\end{equation}
where \(x(t) \in \XO \subset \mathbb{R}^N\) denotes the state within a compact Operational Design Domain (ODD) containing the origin in its interior, \(u(t) \in U_{\mathrm{adm}} := \{u \in \mathbb{R}^m : \|u\|_2 \le u_{\max}\}\) is the norm-bounded control input, \(w(t) \in \mathcal{B}(\varepsilon_{\mathrm{adv}}) := \{w \in \mathbb{R}^{n_w} : \|w\|_2 \le \varepsilon_{\mathrm{adv}}\}\) is a bounded, Lebesgue-measurable exogenous or adversarial disturbance, \(G \in \mathbb{R}^{N\times m}\) and \(D \in \mathbb{R}^{N\times n_w}\) are constant input matrices, \(W_t\) is a standard \(n_W\)-dimensional Wiener process, and \(\Sigma(x) \in \mathbb{R}^{N\times n_W}\) is the diffusion coefficient (\(\Sigma \equiv 0\) in the deterministic setting of Theorem~\ref{thm:main_cbf}). All variables are nondimensionalized (positions, momenta, and field amplitudes scaled by fixed reference values), so that Euclidean norms of vectors mixing generalized coordinates and momenta are meaningful and every constant below is a pure number. Let \(\rho_{\mathrm{ODD}} := \sup\{\rho > 0 : \{x : \|x\|_2 < \rho\} \subseteq \XO\} > 0\) denote the radius of the largest origin-centred Euclidean ball contained in \(\XO\). In the deterministic case, solutions of \eqref{eq:full_system} are understood in the Carath\'eodory sense (absolutely continuous, satisfying the integral equation for the given measurable \(w\) and the piecewise-constant \(u\) of \eqref{eq:multirate_hold}).

\begin{assumption}[Dissipative Linear Port-Hamiltonian Part with Lipschitz Residual]
\label{assump:generic}
The full-order drift decomposes into a linear structure operator and a nonlinear residual, \(F(x) = (J_{\mathrm{full}} - R_{\mathrm{full}})\Pf x + F_{\mathrm{nl}}(x)\) with \(F_{\mathrm{nl}}(0) = 0\), where \(J_{\mathrm{full}} = -J_{\mathrm{full}}^T \in \mathbb{R}^{N \times N}\) is skew-symmetric, \(R_{\mathrm{full}} = R_{\mathrm{full}}^T \succeq 0\) is symmetric positive semi-definite, \(\Pf = \Pf^T \succ 0\) defines the physical energy norm \(\|x\|_{\Pf} := \sqrt{x^T \Pf x}\), \(F_{\mathrm{nl}}\) is locally Lipschitz on \(\mathbb{R}^N\), and \(\|F_{\mathrm{nl}}(x_1) - F_{\mathrm{nl}}(x_2)\|_2 \le L_{F_{\mathrm{nl}}}\|x_1 - x_2\|_2\) for all \(x_1, x_2 \in \XO\). We write \(\Af := (J_{\mathrm{full}} - R_{\mathrm{full}})\Pf\).
\end{assumption}

The decomposition carries no structural content beyond linearity of the port-Hamiltonian part; all nonlinear effects are lumped into \(F_{\mathrm{nl}}\) and enter only through \(L_{F_{\mathrm{nl}}}\) and the suprema defined below.

\paragraph{Reduced coordinates (Petrov--Galerkin).}
Let \(\Phi \in \mathbb{R}^{N \times K}\) (\(K \ll N\)) have full column rank, \(\mathcal{V}_r := \mathrm{Im}(\Phi)\), and define
\begin{equation}
\label{eq:pg_coords}
P_0 := \Phi^T \Pf \Phi \succ 0, \qquad z(x) := P_0^{-1}\Phi^T \Pf\, x \in \mathbb{R}^K, \qquad P_\perp := I_N - \Phi P_0^{-1}\Phi^T \Pf, \qquad x_\perp := P_\perp x .
\end{equation}
Then \(P_\perp^2 = P_\perp\), \(x = \Phi z(x) + x_\perp\) exactly, and \(\Phi^T\Pf x_\perp = \Phi^T\Pf x - P_0 z(x) = 0\) for every \(x \in \mathbb{R}^N\); \(\mathcal{V}_\perp := \mathrm{Im}(P_\perp)\). If \(\Phi\) is an orthonormal eigenbasis with \(\Pf\Phi = \Phi P_0\) (mass-normalized modal coordinates), then \(P_0^{-1}\Phi^T\Pf = \Phi^T\), so \eqref{eq:pg_coords} reduces to \(z = \Phi^T x\) and \(P_\perp = I_N - \Phi\Phi^T\). For second-order Hamiltonian systems (\(x = [q^T, p^T]^T \in \mathbb{R}^{2N_m}\)), \(\Phi = \mathrm{blkdiag}(\Phi_q, \Phi_q)\) with \(\Phi_q^T\Phi_q = I_{K_m}\) and \(K = 2K_m\) even; \(p_r := \Phi_q^T p\) denotes the reduced momenta.

\paragraph{Lyapunov metric and barrier.}
We consider two classes. \emph{Class (D), strictly dissipative:} \(R_{\mathrm{full}} \succ 0\) and \(\Pfe := \Pf\). \emph{Class (H), underdamped second-order Hamiltonian:} \(J_{\mathrm{full}} = \begin{bsmallmatrix} 0 & I_{N_m} \\ -I_{N_m} & 0 \end{bsmallmatrix}\), \(\Pf = \mathrm{blkdiag}(\Omega_{\mathrm{full}}^2, I_{N_m})\), where \(\Omega_{\mathrm{full}}^2 = (\Omega_{\mathrm{full}}^2)^T \succ 0\) is the mass-normalized stiffness matrix with eigenvalues \(\omega_1^2 \le \dots \le \omega_{N_m}^2\) and \(\omega_{\min} := \omega_1\) the smallest \emph{full-order} natural frequency, \(R_{\mathrm{full}} = \mathrm{blkdiag}(0, R_p)\) with momentum damping \(R_p = R_p^T \succeq 0\), and a damping level \(\gamma > 0\) is fixed with
\begin{equation}
\label{eq:eps_choice}
0 < \gamma < 2\omega_{\min}, \qquad \epsilon := \gamma/2, \qquad \Pfe := \begin{bmatrix} \Omega_{\mathrm{full}}^2 & \epsilon I_{N_m} \\ \epsilon I_{N_m} & I_{N_m} \end{bmatrix} \succ 0
\end{equation}
(positive definite because its Schur complement \(\Omega_{\mathrm{full}}^2 - \epsilon^2 I \succeq (\omega_{\min}^2 - \epsilon^2)I \succ 0\)). The same \(\epsilon\) is used for the tail and for the reduced metric (there is no separate \(\epsilon_\perp\)). \(\|x\|_{\Pfe}^2 = x^T\Pfe x\) is a Lyapunov metric, not a physical energy. The barrier is defined on all of \(\mathbb{R}^N\) as
\begin{equation}
\label{eq:h_def}
h(x) := E_{\mathrm{safe}} - x^T \Pfe\, x, \qquad \mathcal{C} := \{x \in \XO : h(x) \ge 0\},
\end{equation}
with a threshold \(E_{\mathrm{safe}} > 0\) satisfying the ODD containment hypothesis
\begin{equation}
\label{eq:H0}
\text{(H0)}\qquad E_{\mathrm{safe}} < \lambda_{\min}(\Pfe)\,\rho_{\mathrm{ODD}}^2 .
\end{equation}

\begin{lemma}[ODD containment]
\label{lem:odd_containment}
Under (H0), \(\mathcal{C} = \{x \in \mathbb{R}^N : x^T\Pfe x \le E_{\mathrm{safe}}\}\), and \(\mathcal{C} \subset \mathrm{int}(\XO)\).
\end{lemma}
\begin{proof}
If \(x^T\Pfe x \le E_{\mathrm{safe}}\), then \(\lambda_{\min}(\Pfe)\|x\|_2^2 \le x^T\Pfe x \le E_{\mathrm{safe}} < \lambda_{\min}(\Pfe)\rho_{\mathrm{ODD}}^2\) (Rayleigh quotient), so \(\|x\|_2 < \rho_{\mathrm{ODD}}\). The open ball \(\{\|x\|_2 < \rho_{\mathrm{ODD}}\}\) is open and contained in \(\XO\), hence contained in \(\mathrm{int}(\XO)\). Thus the ellipsoid lies in \(\mathrm{int}(\XO)\) and coincides with \(\mathcal{C}\).
\end{proof}

\paragraph{Safety and Multi-Rate Timing Specification.}
The digital controller operates at sampling period \(T_s > 0\) with sample instants \(t_k = k T_s\) (\(k \in \mathbb{N}_0\)) and sampled reduced states \(z_k := z(x(t_k))\). The per-axis width \(\DO > 0\) is a given design constant and \(\ZO := [-\DO/2, \DO/2]^K\). The controller uses a dual-rate execution schedule built from a \(b_B\)-bit fixed-point network map \(\Psi_B : \mathbb{R}^K \to U_{\mathrm{adm}}\) whose construction and error certificate are given in Section~\ref{sec:methodology} and Proposition~\ref{prop:twin_crown}:
\begin{enumerate}
    \item \textbf{Fast Feedforward LUT Path (\(\Psi_A\)):} At each sample instant \(t_k\), parallelized FPGA adder-tree projection acquires \(z_k \in \ZO\) within an ultra-short projection sub-interval \(\tau_{\mathrm{proj}} < \tau_A\), leaving \(\tau_A - \tau_{\mathrm{proj}}\) for quantization and table look-up. The quantizer \(Q_{b_A}\) is the mid-tread lattice with \(2^{b_A}+1\) levels \(\{j\Delta_{q,A} : j = -2^{b_A-1}, \dots, 2^{b_A-1}\}\) per axis, step \(\Delta_{q,A} := \DO/2^{b_A}\) (\(b_A \ge 1\)), and half-open bins \([ (j-\tfrac12)\Delta_{q,A}, (j+\tfrac12)\Delta_{q,A})\) (the outermost bins closed at \(\pm\DO/2\)), so that \(\|Q_{b_A}(z) - z\|_\infty \le \Delta_{q,A}/2\) for every \(z \in \ZO\) and the zero bin is the open deadzone \(\mathcal{B}_0 := \{z : \|z\|_\infty < \Delta_{q,A}/2\}\). The table stores \(\Psi_A(z_q) := \Psi_B(z_q)\) for lattice points \(z_q \neq 0\) and \(\Psi_A(0) := 0\) (deadzone; only Path A is dead-zoned), and latches \(u_{A,k} = \Psi_A(Q_{b_A}(z_k))\) at latency \(\tau_A < T_s\). With \(B_{\max}\) address bits, on-chip memory \(S_{\mathrm{bits}}\), and \(b_{\mathrm{out}}\) bits per output component, the address-width and storage conditions are
    \begin{equation}
    \label{eq:lut_capacity}
    K (b_A + 1) \le B_{\max}, \qquad K(b_A + 1) + \lceil \log_2(m\, b_{\mathrm{out}}) \rceil \le \log_2 S_{\mathrm{bits}} ,
    \end{equation}
    since \(\lceil\log_2(2^{b_A}+1)\rceil = b_A + 1\) address bits are needed per axis and the table holds \(2^{K(b_A+1)}\) words of \(m\,b_{\mathrm{out}}\) bits. Any decode latency of the table (e.g., multi-cycle access at large depth) is part of \(\tau_A\).
    \item \textbf{Fine Quantized Neural Path (\(\Psi_B\)):} A \(b_B\)-bit (\(b_B > b_A\)) quantized neural network executes across the multi-layer matrix-vector pipeline and commits a refined command \(u_{B,k} = \Psi_B(z_k)\) at latency \(\tau_B \in (\tau_A, T_s)\). The piecewise-constant control applied to the physical plant over \(t \in [t_k, t_{k+1})\) is
    \begin{equation}
    \label{eq:multirate_hold}
    u(t) = \begin{cases}
    u_{B,k-1}, & t \in [t_k,\, t_k + \tau_A), \\
    u_{A,k},   & t \in [t_k + \tau_A,\, t_k + \tau_B), \\
    u_{B,k},   & t \in [t_k + \tau_B,\, t_k + T_s),
    \end{cases}
    \end{equation}
    where, prior to closed-loop release at \(t_0 = 0\), the plant is held in standby at rest (\(x(t) \equiv x(0)\) for \(t \in [-T_s, 0]\)) and the actuator register is loaded causally with \(u_{B,-1} := \Psi_B(z(x(-T_s))) = \Psi_B(z_0)\).
\end{enumerate}

\section{Structure-Preserving Reduction and Quantized Synthesis}
\label{sec:methodology}

\subsection{Petrov--Galerkin Projection and Spectral Tail Contraction}
Applying \(z(\cdot)\) of \eqref{eq:pg_coords} to \eqref{eq:full_system} (with \(\Sigma \equiv 0\)) and using \(x = \Phi z + x_\perp\) gives the exact reduced dynamics
\begin{equation}
\label{eq:reduced_dynamics}
\dot{z}(t) = \underbrace{\big[A_{r,0} z + F_{\mathrm{nl},r}(z)\big]}_{=: f_r(z)} + \underbrace{P_0^{-1}\Phi^T \Pf G}_{=: g_r} u(t) + \underbrace{P_0^{-1}\Phi^T \Pf D}_{=: d_r} w(t) + \Delta_\perp(z, x_\perp),
\end{equation}
where \(A_{r,0} := (J_0 - R_0)P_0\) is the open-loop reduced operator with
\(J_0 := P_0^{-1}\Phi^T\Pf J_{\mathrm{full}}\Pf\Phi P_0^{-1} = -J_0^T\) and \(R_0 := P_0^{-1}\Phi^T\Pf R_{\mathrm{full}}\Pf\Phi P_0^{-1} = R_0^T \succeq 0\) (so that \(P_0^{-1}\Phi^T\Pf\Af\Phi = (J_0 - R_0)P_0\) identically), \(F_{\mathrm{nl},r}(z) := P_0^{-1}\Phi^T\Pf F_{\mathrm{nl}}(\Phi z)\), and
\[
\Delta_\perp(z, x_\perp) := A_{r\perp} x_\perp + P_0^{-1}\Phi^T\Pf\big[F_{\mathrm{nl}}(\Phi z + x_\perp) - F_{\mathrm{nl}}(\Phi z)\big], \qquad A_{r\perp} := P_0^{-1}\Phi^T \Pf \Af P_\perp .
\]
In the eigenbasis case these reduce to \(J_0 = \Phi^T J_{\mathrm{full}}\Phi\), \(R_0 = \Phi^T R_{\mathrm{full}}\Phi\), \(g_r = \Phi^T G\), \(d_r = \Phi^TD\); in class (H) with orthonormal \(\Phi_q\), \(J_0\) is canonical and \(P_0 = \mathrm{blkdiag}(\Phi_q^T\Omega_{\mathrm{full}}^2\Phi_q, I_{K_m})\). The cross-coupling vanishes, \(A_{r\perp} = 0\), if and only if \(\mathcal{V}_r\) is invariant under the \(\Pf\)-adjoint \(\Pf^{-1}\Af^T\Pf\) (because \(\ker P_\perp^T = \Pf\mathcal{V}_r\) and \(A_{r\perp} = 0 \iff P_\perp^T\Af^T\Pf\Phi = 0\)); when \(\Pf\mathcal{V}_r = \mathcal{V}_r\) this is \(\Af^T\mathcal{V}_r \subseteq \mathcal{V}_r\). Invariance of \(\mathcal{V}_r\) under \(\Af\) itself is neither necessary nor sufficient. \(A_{r\perp} \neq 0\) captures non-commuting advective or elastic cross-coupling.

\begin{assumption}[Exponentially Dissipative Complement in the Unified Metric]
\label{assump:tail_iss}
With \(\Pfe\) as in \eqref{eq:h_def}--\eqref{eq:eps_choice}:
\begin{enumerate}
\item[(H2a)] \emph{Tail \(\Pfe\)-orthogonality:} \(\Phi^T\Pfe P_\perp = 0\). (In class (D) this holds automatically by \eqref{eq:pg_coords}. In class (H) with \(\Phi = \mathrm{blkdiag}(\Phi_q,\Phi_q)\), \(\Phi_q^T\Phi_q = I\), it holds if and only if \(\mathrm{Im}(\Phi_q)\) is \(\Omega_{\mathrm{full}}^2\)-invariant, e.g.\ for any set of mass-normalized eigenmodes; see Lemma~\ref{lem:h2a_char} in Appendix~B.)
\item[(H2b)] \emph{Tail damping (class (H)):} the momentum-damping block restricted to the tail, \(R_{p,\perp} := (I - \Phi_q\Phi_q^T)R_p(I - \Phi_q\Phi_q^T)\), satisfies \(v^T R_{p,\perp} v > \epsilon\|v\|_2^2\) for all nonzero \(v \in \mathrm{Im}(I - \Phi_q\Phi_q^T)\). (This is necessary for \eqref{eq:tail_dissipation}; general SPD \(R_p\) violating it yields an indefinite tail dissipation form.)
\item[(H2c)] \emph{Spectral gap:} there is \(\lambda_{K+1} > 0\) with
\begin{equation}
\label{eq:tail_dissipation}
-x_\perp^T \Big(\Pfe \Af + \Af^T \Pfe\Big) x_\perp \ge 2\,\lambda_{K+1} \|x_\perp\|_{\Pfe}^2 \quad \forall x_\perp \in \mathcal{V}_\perp ,
\end{equation}
\end{enumerate}
with uniformly bounded tail excitation over the ODD
\begin{equation}
\label{eq:M_perp}
M_\perp := \sup_{(x,u,w) \in \XO \times U_{\mathrm{adm}} \times \mathcal{B}(\varepsilon_{\mathrm{adv}})} \big\|P_\perp\big[\Af\Phi z(x) + F_{\mathrm{nl}}(x) + Gu + Dw\big]\big\|_{\Pfe} < \infty, \qquad z(x) \text{ as in \eqref{eq:pg_coords}}.
\end{equation}
\end{assumption}
\(M_\perp\) is finite because \(\XO\) is compact and the bracket is continuous. Only upper bounds are used below, so \(M_\perp\) may be replaced by any larger number; we take \(M_\perp > 0\) without loss of generality (if the supremum vanishes, any positive value for which the hypotheses of Theorem~\ref{thm:main_cbf} hold may be used).

\begin{remark}[Tail rates for the two benchmark classes]
\label{rem:tail_rates}
(i) \emph{Uniformly damped lattice} (class (H), \(R_p = \gamma I\), \(\mathrm{Im}\Phi_q\) spanned by eigenmodes): \(\Af\) and \(\Pfe\) decouple into \(2\times2\) modal blocks, and the largest admissible rate of mode \(i\) is the pencil rate \(r(\epsilon,\omega_i) = \frac{\gamma}{2} - \frac{\omega_i|\gamma - 2\epsilon|}{2\sqrt{\omega_i^2 - \epsilon^2}} \le \frac{\gamma}{2}\) (Lemma~\ref{lem:pencil_rate}); hence \(\lambda_{K+1} := \min_{i > K_m} r(\epsilon, \omega_i)\), which equals \(\gamma/2\) for every tail mode at \(\epsilon = \gamma/2\) (and (H2b) holds since \(\gamma > \gamma/2\)). The tail rate does \emph{not} grow with \(K\); truncation enters only through \(M_\perp\) and \(\Lperp\). (ii) \emph{Reaction--diffusion--advection} \(\partial_t x = \nu\partial_{\xi\xi}x - c_{\mathrm{adv}}\partial_\xi x - \alpha_0 x + (\text{nonlinear})\) on \((0,\ell)\) with Dirichlet conditions (class (D), \(\Pf = I\) in \(L^2\), advection skew): with \(\Phi\) the first \(K\) sine modes, \(\lambda_{K+1} = \alpha_0 + \nu\big((K+1)\pi/\ell\big)^2\) for the continuum and \(\lambda_{K+1} = \alpha_0 + \nu\frac{4}{h^2}\sin^2\frac{(K+1)\pi h}{2\ell}\) for the central finite-difference semi-discretization with mesh \(h = \ell/(N+1)\) (exact discrete sine eigenvalues; the FD value is smaller). Both require \(\lambda_{K+1} > 0\), which may fail for \(\alpha_0 < 0\); \(\alpha_0\) is retained. See Remark~\ref{rem:rda_cutoff}.
\end{remark}

\begin{lemma}[Quadratic splitting]
\label{lem:quad_split}
Under (H2a), for every \(x \in \mathbb{R}^N\) with \(z = z(x)\), \(x_\perp = P_\perp x\): \(\Phi^T\Pfe x_\perp = 0\),
\(\|x\|_{\Pfe}^2 = z^T P z + \|x_\perp\|_{\Pfe}^2\), and
\begin{equation}
\label{eq:quad_split}
h(\Phi z + x_\perp) = h(\Phi z) - \|x_\perp\|_{\Pfe}^2, \qquad h(\Phi z) = E_{\mathrm{safe}} - z^T P z, \qquad P := \Phi^T\Pfe\Phi .
\end{equation}
\end{lemma}
\begin{proof}
\(\Phi^T\Pfe x_\perp = \Phi^T\Pfe P_\perp x = 0\) by (H2a). Expanding,
\[
(\Phi z + x_\perp)^T\Pfe(\Phi z + x_\perp) = z^T\Phi^T\Pfe\Phi z + 2z^T(\Phi^T\Pfe x_\perp) + x_\perp^T\Pfe x_\perp = z^TPz + \|x_\perp\|_{\Pfe}^2;
\]
subtract from \(E_{\mathrm{safe}}\).
\end{proof}

The reduced metric is \(P = P_0\) in class (D) and \(P = P_\epsilon := \begin{bsmallmatrix} \Omega_r^2 & \epsilon I \\ \epsilon I & I \end{bsmallmatrix}\) in class (H) with eigenmode \(\Phi_q\) (\(\Omega_r = \mathrm{diag}(\omega_1,\dots,\omega_{K_m})\)). Under (H0) and Lemma~\ref{lem:quad_split}, \(x \in \mathcal{C}\) implies \(\Phi z(x) \in \mathcal{C}\) (since \(\|\Phi z\|_{\Pfe}^2 \le \|x\|_{\Pfe}^2\)). We define the \(\Pfe\)-norm tail coupling constant \(\Lperp\) as the smallest constant such that
\begin{equation}
\label{eq:Lperp_def}
\big\|P_0^{-1}\Phi^T\Pf\big[\Af x_\perp + F_{\mathrm{nl}}(\Phi z + x_\perp) - F_{\mathrm{nl}}(\Phi z)\big]\big\|_2 = \|\Delta_\perp(z,x_\perp)\|_2 \le \Lperp\,\|x_\perp\|_{\Pfe}
\end{equation}
for all \(x = \Phi z + x_\perp \in \XO\) with \(\Phi z \in \XO\). It is finite: \(\Lperp \le \|A_{r\perp}\Pfe^{-1/2}\|_2 + \|P_0^{-1}\Phi^T\Pf\|_2\,L_{F_{\mathrm{nl}}}\,\lambda_{\min}(\Pfe)^{-1/2}\) (using \(A_{r\perp}x_\perp = P_0^{-1}\Phi^T\Pf\Af x_\perp\) because \(P_\perp x_\perp = x_\perp\), and Assumption~\ref{assump:generic}); the sharp constant, not this crude bound, is what enters the certificate. Finally, the \(\Pfe\)-norm tail radius is
\begin{equation}
\label{eq:eps_basis_P}
\epsP := M_\perp/\lambda_{K+1} .
\end{equation}

\subsection{Isometric GroupSort-2 Teacher and Quantized Student}
\label{sec:teacher}
The continuous network \(\mathcal{N} : \mathbb{R}^m \to \mathbb{R}^m\) acts on the port output \(v := g_r^T P z \in \mathbb{R}^m\). It is an origin-preserving (\(b_\ell = 0\)) \(L\)-layer network (\(L \ge 2\)) of hidden width \(H\) (even, \(H/2 \ge m\)) with pairwise \(\mathrm{GroupSort\text{-}2}\) (\(\mathrm{MaxMin}\)) activations implemented via the exact ReLU routing identity
\begin{equation}
\label{eq:groupsort_relu}
\sigma_{\mathrm{GS2}}(y_{2i-1}, y_{2i}) = \begin{bmatrix} \max(y_{2i-1}, y_{2i}) \\ \min(y_{2i-1}, y_{2i}) \end{bmatrix} = \begin{bmatrix} y_{2i} + \mathrm{ReLU}(y_{2i-1} - y_{2i}) \\ y_{2i-1} - \mathrm{ReLU}(y_{2i-1} - y_{2i}) \end{bmatrix}.
\end{equation}
\(\sigma_{\mathrm{GS2}}\) is norm-preserving (\(\|\sigma_{\mathrm{GS2}}(y)\|_2 = \|y\|_2\)) and non-expansive (\(\|\sigma_{\mathrm{GS2}}(y) - \sigma_{\mathrm{GS2}}(\tilde{y})\|_2 \le \|y - \tilde{y}\|_2\)). Let \(U_{\mathrm{sum}} := \frac{1}{\sqrt2}\sum_{i=1}^{H/2}(e_{2i-1} + e_{2i})e_i^T\) and \(U_{\mathrm{diff}} := \frac{1}{\sqrt2}\sum_{i=1}^{H/2}(e_{2i-1} - e_{2i})e_i^T \in \mathbb{R}^{H\times H/2}\), whose ranges are exactly \(\mathrm{span}\{e_{2i-1} + e_{2i}\}\) and \(\mathrm{span}\{e_{2i-1} - e_{2i}\}\), aligned with the GS2 pairing (\(U_{\mathrm{sum}}^T U_{\mathrm{sum}} = U_{\mathrm{diff}}^T U_{\mathrm{diff}} = I_{H/2}\), \(U_{\mathrm{sum}}^T U_{\mathrm{diff}} = 0\), \([U_{\mathrm{sum}}\ U_{\mathrm{diff}}] \in O(H)\)). Fix a mixing angle \(\theta_{\mathrm{GS}} \in (0, \pi/4)\), \(Q_{\mathrm{mix}} \in O(H/2)\), a semi-orthogonal input matrix \(W_{\mathrm{in}} \in \mathbb{R}^{(H/2)\times m}\) (\(W_{\mathrm{in}}^T W_{\mathrm{in}} = I_m\)), and set
\begin{equation}
\label{eq:teacher_weights}
\begin{aligned}
W_1 &= \cos(\theta_{\mathrm{GS}})U_{\mathrm{sum}}W_{\mathrm{in}} + \sin(\theta_{\mathrm{GS}})U_{\mathrm{diff}}Q_{\mathrm{mix}}W_{\mathrm{in}} \in \mathbb{R}^{H\times m}, \qquad W_1^TW_1 = I_m,\\
W_\ell &= U_{\mathrm{sum}}Q_{\mathrm{sum},\ell}U_{\mathrm{sum}}^T + U_{\mathrm{diff}}Q_{\mathrm{diff},\ell}U_{\mathrm{diff}}^T \in O(H) \quad (2 \le \ell \le L-1,\ Q_{\mathrm{sum},\ell}, Q_{\mathrm{diff},\ell} \in O(H/2)),\\
W_L &= -L_{\mathcal{T}}(W_{L-1}\cdots W_1)^T \in \mathbb{R}^{m \times H}, \qquad \mathcal{N}(v) := W_L\,\sigma_{\mathrm{GS2}}(W_{L-1}\cdots\sigma_{\mathrm{GS2}}(W_1 v)).
\end{aligned}
\end{equation}
Hence \(\|W_\ell\|_2 \le 1\) for \(\ell < L\), \(\|W_L\|_2 = L_{\mathcal{T}}\), \(\mathcal{N}(0) = 0\), and \(\mathcal{N}\) is \(L_{\mathcal{T}}\)-Lipschitz in \(\ell_2\). The Teacher and the quantized Student are
\begin{equation}
\label{eq:teacher_def}
\Tt(z) := \Pi_U\big(\mathcal{N}(g_r^TPz)\big), \qquad \Psi_B(z) := \Pi_U\big(\mathcal{N}_q(g_r^TPz)\big), \qquad \Pi_U(a) := a\min\!\big(1, u_{\max}/\|a\|_2\big),
\end{equation}
where \(\mathcal{N}_q\) is the \(b_B\)-bit fixed-point realization of \(\mathcal{N}\) specified in Proposition~\ref{prop:twin_crown} and \(\Pi_U\) is the (radial = metric) projection onto \(U_{\mathrm{adm}}\), which satisfies \(\|\Pi_U a - \Pi_U b\|_2 \le \|a - b\|_2\). Consequently \(\Tt(0) = 0\), \(\Tt\) and \(\Psi_B\) take values in \(U_{\mathrm{adm}}\), every Lipschitz constant and every bound on \(\|\Psi_B - \Tt\|_2\) obtained for the unprojected maps carries over, and \(\Tt\) is \(\LTz\)-Lipschitz in \(z\) with
\begin{equation}
\label{eq:LTz}
\LTz := L_{\mathcal{T}}\,\|g_r^T P\|_2 .
\end{equation}
We assume the port Gram matrix \(G_{\mathrm{port}} := g_r^TP^2g_r \succ 0\) (full column rank of \(g_r\)), so the port map \(z \mapsto g_r^TPz\) is onto \(\mathbb{R}^m\) and its null space has dimension exactly \(K-m\), and the saturation hypothesis
\begin{equation}
\label{eq:sat_cond}
\text{(H-sat)}\qquad \LTz\,\sqrt{K}\,\DO/2 \le u_{\max},
\end{equation}
under which \(\|\mathcal{N}(g_r^TPz)\|_2 \le \LTz\|z\|_2 \le \LTz\sqrt{K}\DO/2 \le u_{\max}\) on \(\ZO\), so \(\Pi_U\) acts as the identity on the image \(\mathcal{N}(g_r^TP\,\ZO)\). (Without (H-sat), Lemma~\ref{lem:sector} still holds with \(\eta_{\mathcal{T}}\) replaced by \(\eta_{\mathcal{T}}\min(1, u_{\max}/(L_{\mathcal{T}}\|v\|_2))\).)

\begin{lemma}[Sector alignment; proof in Appendix~C]
\label{lem:sector}
For the construction \eqref{eq:teacher_weights}--\eqref{eq:teacher_def} under (H-sat), every \(z \in \ZO\) satisfies
\(-z^T P g_r \Tt(z) \ge \eta_{\mathcal{T}}\|g_r^T P z\|_2^2\) with \(\eta_{\mathcal{T}} = L_{\mathcal{T}}\cos(2\theta_{\mathrm{GS}}) > 0\); in particular \(z^TPg_r\Tt(z) \le 0\).
\end{lemma}

While the linear sum channel \(U_{\mathrm{sum}}\) enforces this baseline \(P\)-dissipative floor, the rectified difference channel \(U_{\mathrm{diff}}\) splits the port space by first-layer comparator signs into at most \(2^{H/2}\) polyhedral cones (all linear regions of \(\mathcal{N}\) are cones because \(\mathcal{N}\) is positively homogeneous), providing piecewise-linear performance shaping and actuator allocation beyond a static linear gain \(u = -k_{\mathrm{lin}}\,g_r^T P z\) (which avoids saturation on \(\ZO\) only if \(k_{\mathrm{lin}}\|g_r^TP\|_2\sqrt{K}\DO/2 \le u_{\max}\)). The control term vanishes on the \((K-m)\)-dimensional subspace \(\{g_r^TPz = 0\}\), which meets every level set of \(h(\Phi z)\); there safety must be carried by passive dissipation (condition \eqref{eq:null_space_cond} below).

\subsection{Coupled Twin-Network Verification and Structure-Preserving Contraction}

\begin{proposition}[Coupled Twin-Network CROWN Verification Bound]
\label{prop:twin_crown}
Let \(\mathcal{N}\) be the network \eqref{eq:teacher_weights} with layer widths \(H_0 := m\), \(H_\ell := H\) (\(1 \le \ell \le L-1\)), \(H_L := m\), weights satisfying \(\|W_\ell\|_2 \le 1\) (\(\ell < L\)), \(\|W_L\|_2 = L_{\mathcal{T}}\), and zero biases. Let \(\mathcal{N}_q\) be its fixed-point realization with weights \(W_\ell^q\) obtained by per-tensor quantization with scale \(s_\ell > 0\) such that \(|[W_\ell^q - W_\ell]_{ij}| \le s_\ell 2^{-b_B}\), hence
\(\|W_\ell^q - W_\ell\|_2 \le \Delta_{W,\ell} \le \sqrt{H_\ell H_{\ell-1}}\,s_\ell\,2^{-b_B}\), biases \(b_\ell^q\) with \(\|b_\ell^q\|_2 \le \Delta_{b,\ell}\), and rounding of the input, accumulators and activations modelled as injections \(e_0 \in [-\rho_0,\rho_0]^{H_0}\) (input \(v + e_0\)) and \(e_\ell \in [-\rho_\ell, \rho_\ell]^{H_\ell}\) added to each quantized pre-activation, \(y_\ell^q = W_\ell^q\sigma_{\mathrm{GS2}}(y_{\ell-1}^q) + b_\ell^q + e_\ell\) (with \(\sigma_{\mathrm{GS2}}(y_0^q) := v + e_0\)). Write \(y_\ell, y_\ell^q\) for the pre-activations as functions of \(z\) (through \(v = g_r^TPz\)) and of the injections. Over any sub-hyperrectangle \(\mathcal{Z}_s = [\underline{z}_s, \overline{z}_s] \subset \ZO\), let \(d_{\ell,i}(z) := y_{\ell,2i-1}(z) - y_{\ell,2i}(z) \in [\underline{d}_{\ell,i}, \overline{d}_{\ell,i}]\) and \(d_{\ell,i}^q(z) := y_{\ell,2i-1}^q(z) - y_{\ell,2i}^q(z) \in [\underline{d}_{\ell,i}^q, \overline{d}_{\ell,i}^q]\) denote the layer-\(\ell\) comparators on the shared ReLU routing graph \eqref{eq:groupsort_relu}, with perturbation \(\delta d_{\ell,i}(z) := d_{\ell,i}^q(z) - d_{\ell,i}(z) \in [\underline{\delta}_{\ell,i}, \overline{\delta}_{\ell,i}]\), all intervals being produced by the same backward pass over \(\mathcal{Z}_s\) and the injection boxes. Let \(\mathcal{I}_\ell^+(\mathcal{Z}_s) := \{i : \underline{d}_{\ell,i} \ge 0,\ \underline{d}^q_{\ell,i} \ge 0\}\), \(\mathcal{I}_\ell^-(\mathcal{Z}_s) := \{i : \overline{d}_{\ell,i} \le 0,\ \overline{d}^q_{\ell,i} \le 0\}\), and \(\mathcal{I}_\ell^*(\mathcal{Z}_s) := \{1,\dots,H/2\}\setminus(\mathcal{I}_\ell^+ \cup \mathcal{I}_\ell^-)\) (which contains every comparator whose sorting order can flip under quantization). Then:
\begin{enumerate}
\item[(i)] In the coupled difference graph \(\Delta_{\mathrm{net}}(z) := \mathcal{N}_q(g_r^TPz) - \mathcal{N}(g_r^TPz)\), every comparator difference \(\nu_{\ell,i}(z) := \mathrm{ReLU}(d_{\ell,i}^q(z)) - \mathrm{ReLU}(d_{\ell,i}(z))\) admits the sound two-sided affine relaxation in \(\delta d_{\ell,i}(z)\)
\begin{equation}
\label{eq:twin_affine_relax}
\alpha_{\ell,i,s}^L\,\delta d_{\ell,i}(z) + \beta_{\ell,i,s}^L \le \nu_{\ell,i}(z) \le \alpha_{\ell,i,s}^U\,\delta d_{\ell,i}(z) + \beta_{\ell,i,s}^U,
\end{equation}
with non-negative slopes \(\alpha_{\ell,i,s}^L, \alpha_{\ell,i,s}^U \in [0, 1]\), exact (zero-slack) coefficients \(\alpha^L = \alpha^U = 1\), \(\beta^L = \beta^U = 0\) on \(\mathcal{I}_\ell^+(\mathcal{Z}_s)\) and \(\alpha^L = \alpha^U = 0\), \(\beta^L = \beta^U = 0\) on \(\mathcal{I}_\ell^-(\mathcal{Z}_s)\), and, for \(i \in \mathcal{I}_\ell^*(\mathcal{Z}_s)\), the coefficients (all depending on \(s\) through \([\underline{\delta}_{\ell,i}, \overline{\delta}_{\ell,i}]\))
\begin{equation}
\label{eq:twin_slopes}
\resizebox{0.89\columnwidth}{!}{$\displaystyle
(\alpha_{\ell,i,s}^L, \beta_{\ell,i,s}^L, \alpha_{\ell,i,s}^U, \beta_{\ell,i,s}^U) = \begin{cases}
(0,\, 0,\, 1,\, 0), & \text{if } \underline{\delta}_{\ell,i} \ge 0, \\
(1,\, 0,\, 0,\, 0), & \text{if } \overline{\delta}_{\ell,i} \le 0, \\
\left(\dfrac{-\underline{\delta}_{\ell,i}}{\overline{\delta}_{\ell,i} - \underline{\delta}_{\ell,i}},\, \dfrac{\underline{\delta}_{\ell,i}\,\overline{\delta}_{\ell,i}}{\overline{\delta}_{\ell,i} - \underline{\delta}_{\ell,i}},\, \dfrac{\overline{\delta}_{\ell,i}}{\overline{\delta}_{\ell,i} - \underline{\delta}_{\ell,i}},\, \dfrac{-\underline{\delta}_{\ell,i}\,\overline{\delta}_{\ell,i}}{\overline{\delta}_{\ell,i} - \underline{\delta}_{\ell,i}}\right), & \text{if } \underline{\delta}_{\ell,i} < 0 < \overline{\delta}_{\ell,i}.
\end{cases}$}
\end{equation}
The pointwise envelope \(\min(0,\delta d) \le \nu \le \max(0,\delta d)\) underlying \eqref{eq:twin_slopes} is attained (upper side at \(d_{\ell,i} = 0\), lower side at \(d_{\ell,i} = -\delta d_{\ell,i}\)); the secant chord is exact at the endpoints of \([\underline{\delta}_{\ell,i}, \overline{\delta}_{\ell,i}]\).
\item[(ii)] Propagating \eqref{eq:twin_affine_relax} by backward linear substitution over \(\mathcal{Z}_s\) (for the upper bound of each output row, choosing \((\alpha^U, \beta^U)\) for comparator \(i\) where its accumulated coefficient \(c_{\ell,i} := [\Lambda W^q_{\ell+1}]_{\cdot,2i-1} - [\Lambda W^q_{\ell+1}]_{\cdot,2i}\) is non-negative and \((\alpha^L,\beta^L)\) otherwise, and the reverse for the lower bound; \(\Lambda\) is the backward coefficient matrix accumulated from the output) yields component-wise bounds \([\Delta_{\mathrm{net}}(z)]_j \in [\underline{D}_{s,j}, \overline{D}_{s,j}]\), \(j = 1,\dots,m\), and for any fixed partition \(\ZO = \bigcup_{s=1}^{M_{\mathrm{sub}}} \mathcal{Z}_s\) of the \emph{whole} domain into \(M_{\mathrm{sub}}\) boxes,
\begin{equation}
\label{eq:twin_crown_bound}
\sup_{z \in \ZO} \|\Psi_B(z) - \Tt(z)\|_2 \le \sup_{z \in \ZO} \|\Delta_{\mathrm{net}}(z)\|_2 \le \max_{1 \le s \le M_{\mathrm{sub}}} \left( \sum_{j=1}^m \max\!\big(|\underline{D}_{s,j}|^2,\, |\overline{D}_{s,j}|^2\big) \right)^{1/2} =: \eC .
\end{equation}
\item[(iii)] (Explicit constant.) With \(Y := \|g_r^TP\|_2\sqrt{K}\DO/2\) (so \(\|v\|_2 \le Y\) on \(\ZO\)) and the convention \(\Delta_{W,0} = \Delta_{b,0} = 0\),
\begin{equation}
\label{eq:crown_explicit}
\sup_{z \in \ZO}\|\Delta_{\mathrm{net}}(z)\|_2 \le E_q(b_B) := \sum_{\ell=0}^{L}\Big(\Delta_{W,\ell}\,Y + \Delta_{b,\ell} + \sqrt{H_\ell}\,\rho_\ell\Big)\prod_{k>\ell}\|W_k^q\|_2,
\qquad
\eC \le C_{\mathrm{relax}}\,E_q(b_B),
\end{equation}
where \(\prod_{k>\ell}\|W_k^q\|_2 \le (L_{\mathcal{T}} + \Delta_{W,L})\prod_{k<L}(1 + \Delta_{W,k})\) for \(\ell < L\) (the empty product for \(\ell = L\) is \(1\)), and \(C_{\mathrm{relax}} := \eC/E_q(b_B) \in (0,\infty)\) is the (computable) relaxation factor of the architecture and partition; \(C_{\mathrm{relax}} \le \sqrt{m}\) whenever every comparator lies in \(\mathcal{I}^+_\ell \cup \mathcal{I}^-_\ell\) on every box. Since both \(\eC\) and \(E_q(b_B)\) are sound, \(\min(\eC, E_q(b_B))\) may be used in place of \(\eC\) throughout.
\item[(iv)] (Rate.) If \(\Delta_{b,\ell}, \rho_\ell \le c\,2^{-b_B}\) for a constant \(c\), then for fixed architecture, teacher weights, scales \(s_\ell\) and partition, \(E_q(b_B) = O(2^{-b_B})\) and \(\eC = O(2^{-b_B})\).
\end{enumerate}
\end{proposition}

\begin{proposition}[Structure-Preserving Contraction via \(P\)-Weighted Matrix Measure]
\label{prop:dahlquist}
Consider the linearized reduced error dynamics \(\dot{e} = A_{r,0} e\) with \(A_{r,0} = (J_0 - R_0)P_0\), where \(J_0 = -J_0^T\), \(R_0 = R_0^T \succeq 0\), and \(P_0 = P_0^T \succ 0\).
\begin{enumerate}
    \item \textbf{Strictly Dissipative Case (\(R_0 \succ 0\)):} In the energy metric \(\|e\|_{P_0} := \sqrt{e^T P_0 e}\), the conservative skew-symmetric term vanishes identically (\((P_0 e)^T J_0 (P_0 e) = 0\) for all \(e \in \mathbb{R}^K\)), and the \(P_0\)-weighted Dahlquist matrix measure satisfies
    \begin{equation}
    \label{eq:mu_p0}
    \mu_{2,P_0}(A_{r,0}) := \sup_{e \neq 0} \frac{e^T (P_0 A_{r,0} + A_{r,0}^T P_0) e}{2\,e^T P_0 e} \le -\lambda_{\min}(R_0)\,\lambda_{\min}(P_0) =: -\mu_R < 0.
    \end{equation}
    \item \textbf{Underdamped Hamiltonian Case (\(R_0 = \mathrm{blkdiag}(0, \gamma I_{K_m})\), \(\gamma > 0\)):} For canonical second-order systems with \(J_0 = \begin{bsmallmatrix} 0 & I \\ -I & 0 \end{bsmallmatrix}\) and \(P_0 = \mathrm{blkdiag}(\Omega_r^2, I_{K_m})\) (\(\Omega_r = \mathrm{diag}(\omega_1, \dots, \omega_{K_m}) \succ 0\)), the cross-weighted symmetric metric \(P_\epsilon := \begin{bsmallmatrix} \Omega_r^2 & \epsilon I \\ \epsilon I & I \end{bsmallmatrix}\) satisfies \(P_\epsilon \succ 0\) and \(Q_\epsilon := -(P_\epsilon A_{r,0} + A_{r,0}^TP_\epsilon) \succ 0\) for every \(\epsilon \in \big(0,\, \frac{4\gamma \omega_{\min}^2}{\gamma^2 + 4\omega_{\min}^2}\big)\). The choice \(\epsilon = \gamma/2\) lies in this interval if and only if \(\gamma < 2\omega_{\min}\); for it, \(Q_{\gamma/2} = \gamma P_{\gamma/2}\) and
    \begin{equation}
    \label{eq:mu_pe}
    \mu_{2,P_{\gamma/2}}(A_{r,0}) = -\gamma/2 =: -\mu_R, \qquad \|e(t)\|_{P_{\gamma/2}} = e^{-\gamma t/2}\|e(0)\|_{P_{\gamma/2}} .
    \end{equation}
    Moreover \((1 - \epsilon/\omega_{\min})P_0 \preceq P_\epsilon \preceq (1 + \epsilon/\omega_{\min})P_0\).
\end{enumerate}
Consequently, in both cases, an ellipsoidal error tube \(\mathcal{E}_P(r) = \{e \in \mathbb{R}^K : \|e\|_P \le r\}\) contracts strictly at rate \(e^{-\mu_R \Delta t} < 1\), whereas axis-aligned interval (\(\ell_\infty\)) propagation incurs artificial conservative wrapping growth \(\mu_\infty(A_{r,0}) = \max_i\big((A_{r,0})_{ii} + \sum_{j \neq i} |(A_{r,0})_{ij}|\big) > 0\) whenever off-diagonal skew-symmetric couplings dominate diagonal damping.
\end{proposition}

\subsection{Sampled-Data Constants and the Tightened Reduced Barrier}
All reduced constants are evaluated on the \emph{untightened} reduced safe set
\begin{equation}
\label{eq:Cr0}
\Crz := \{z \in \ZO : h(\Phi z) \ge 0\} = \{z \in \mathbb{R}^K : z^TPz \le E_{\mathrm{safe}}\} \subset \mathrm{int}(\ZO),
\end{equation}
where the equality and the inclusion follow from the containment condition \(E_{\mathrm{safe}} < \frac{1}{4}\DO^2\lambda_{\min}(P)\), assumed together with (H0): \(z^TPz \le E_{\mathrm{safe}}\) gives \(\|z\|_\infty \le \|z\|_2 \le (E_{\mathrm{safe}}/\lambda_{\min}(P))^{1/2} < \DO/2\). Since \(\nabla_z h(\Phi z) = \Phi^T\nabla_x h(\Phi z) = -2Pz\) does not depend on any constant level shift, we define on \(\Crz\) the Lie derivatives \(\mathcal{L}_{f_r}\tilde{h}_r(z) := -2z^TPf_r(z)\), \(\mathcal{L}_{g_r}\tilde{h}_r(z) := -2z^TPg_r\), \(\mathcal{L}_{d_r}\tilde{h}_r(z) := -2z^TPd_r\) (valid for any \(\tilde h_r\) of the form \(h(\Phi z) - \text{const}\)), and the constants
\begin{equation}
\label{eq:reduced_constants}
\begin{aligned}
&L_h := \sup_{z \in \Crz}\|2Pz\|_2 = 2\sqrt{\lambda_{\max}(P)E_{\mathrm{safe}}}, \quad \overline{g}_h := \sup_{z \in \Crz}\|\mathcal{L}_{g_r}\tilde{h}_r(z)\|_2, \quad \overline{d}_h := \sup_{z \in \Crz}\|\mathcal{L}_{d_r}\tilde{h}_r(z)\|_2,\\
&M_z := \sup_{(z,u,w) \in \Crz \times U_{\mathrm{adm}} \times \mathcal{B}(\varepsilon_{\mathrm{adv}})} \|f_r(z) + g_r u + d_r w\|_2 + \Lperp\,\epsP,
\end{aligned}
\end{equation}
and \(L_{\mathcal{L}_{f_r}\tilde{h}_r}\), \(L_{\mathcal{L}_{g_r}\tilde{h}_r}\) the Lipschitz constants of \(\mathcal{L}_{f_r}\tilde{h}_r\) and \(\mathcal{L}_{g_r}\tilde{h}_r\) on \(\Crz\) (the latter equals \(2\|g_r^TP\|_2\)). None of these depends on \(\Delta h_{\mathrm{dip}}\), \(h_{\mathrm{act}}\) or the shell, and \(\epsP\) depends only on \(M_\perp\) and \(\lambda_{K+1}\), so the definitions are acyclic. The \(\Pfe\)-norm form of the tail tolerance removes \(\lambda_{\min}(\Pfe)\) entirely from the certificate.

\paragraph{Path-A error and co-design window.}
By Proposition~\ref{prop:twin_crown} and \(\LTz\)-Lipschitz continuity of \(\Tt\), the Path-B QNN \(\Psi_B(z)\) and Path-A LUT \(\Psi_A(Q_{b_A}(z))\) satisfy \(\|\Psi_B(z) - \Tt(z)\|_2 \le \eC\) on \(\ZO\) and, for \(Q_{b_A}(z) \neq 0\),
\begin{equation}
\label{eq:eps_A_triangle}
\begin{aligned}
\|\Psi_A(Q_{b_A}(z)) - \Tt(z)\|_2 &\le \|\Psi_B(Q_{b_A}(z)) - \Tt(Q_{b_A}(z))\|_2 + \|\Tt(Q_{b_A}(z)) - \Tt(z)\|_2 \\
&\le \eC + \LTz\frac{\sqrt{K}\DO}{2^{b_A+1}} =: \eA,
\end{aligned}
\end{equation}
while for \(Q_{b_A}(z) = 0\) (deadzone, \(\Psi_A = 0\)) \(\|0 - \Tt(z)\|_2 = \|\Tt(0) - \Tt(z)\|_2 \le \LTz\|z\|_2 \le \LTz\sqrt{K}\DO/2^{b_A+1} \le \eA\). Define \(\dul := \LTz M_z T_s\). The co-design window is
\begin{equation}
\label{eq:codesign_window}
\left\lfloor \log_2\!\left(\frac{\sqrt{K}\DO}{2 M_z T_s}\right) \right\rfloor + 1 \;\le\; b_A \;\le\; \left\lfloor \frac{B_{\max}}{K} \right\rfloor - 1 ,
\end{equation}
together with the storage condition in \eqref{eq:lut_capacity}. The upper bound is the address-width condition of \eqref{eq:lut_capacity}. The lower bound is a design choice that is not used in the proof of Theorem~\ref{thm:main_cbf} (which holds for every \(b_A \ge 1\)); it guarantees \(\eC < \eA < \eC + \dul\): indeed \(2^{b_A} \ge 2^{\lfloor \log_2 y\rfloor + 1} > y\) with \(y = \sqrt{K}\DO/(2M_zT_s)\) gives \(\LTz\sqrt{K}\DO/2^{b_A+1} < \LTz M_zT_s = \dul\), and \(\LTz\sqrt K\DO > 0\) gives the left inequality.

\paragraph{Exponential weights, hold factor, dip buffer and margin.}
For sampling period \(T_s\), delays \(0 < \tau_A < \tau_B < T_s\) and a linear barrier decay rate \(\gamma_0 > 0\) (the rate used in the certificate \eqref{eq:teacher_cbf} below), define the normalized exponential sub-interval weights
\begin{equation}
\label{eq:exp_weights}
w_1(\gamma_0) := \frac{e^{\gamma_0 \tau_A} - 1}{e^{\gamma_0 T_s} - 1}, \quad w_2(\gamma_0) := \frac{e^{\gamma_0 \tau_B} - e^{\gamma_0 \tau_A}}{e^{\gamma_0 T_s} - 1}, \quad w_3(\gamma_0) := \frac{e^{\gamma_0 T_s} - e^{\gamma_0 \tau_B}}{e^{\gamma_0 T_s} - 1},
\end{equation}
the dimensionless hold-integral factor \(\varphi(\xi) := \frac{(\xi - 1)e^\xi + 1}{\xi(e^\xi - 1)} \in (\frac{1}{2}, 1)\),\footnote{For implementation near \(\xi \to 0\) use the series \(\varphi(\xi) = \tfrac12 + \tfrac{\xi}{12} - \tfrac{\xi^3}{720} + O(\xi^5)\) to avoid cancellation in floating point.} the hold-rate constant \(L_{\mathrm{hold}} := (L_{\mathcal{L}_{f_r}\tilde{h}_r} + L_{\mathcal{L}_{g_r}\tilde{h}_r} u_{\max}\) \(+ \gamma_0 L_h) M_z\), the inter-sample continuous-time dip buffer
\begin{equation}
\label{eq:delta_dip}
\Delta h_{\mathrm{dip}} := \overline{g}_h \big(\varepsilon_A(b_A) - \varepsilon_{\mathrm{CROWN}}(b_B)\big)(\tau_B - \tau_A) + \overline{g}_h\,\Delta u_{\mathrm{lag}}\,\tau_A + \frac{1}{2}L_{\mathrm{hold}} T_s^2,
\end{equation}
and the multi-rate sampled-data robustness margin
\begin{equation}
\label{eq:delta_total_exact}
\begin{aligned}
\delta_{\mathrm{total}}(K, b_A, b_B, T_s) :={}& \underbrace{L_h \Lperp\,\epsP}_{\text{ROM Tail Coupling}} + \underbrace{\overline{g}_h \Big[ w_2(\gamma_0)\,\varepsilon_A(b_A) + \big(1 - w_2(\gamma_0)\big)\varepsilon_{\mathrm{CROWN}}(b_B) \Big]}_{\text{Exponentially Weighted Multi-Rate Quantization Error}} \\
&+ \underbrace{w_1(\gamma_0)\,\overline{g}_h\,\Delta u_{\mathrm{lag}}}_{\text{Pre-Arrival Lag Rate}} + \underbrace{\overline{d}_h\,\varepsilon_{\mathrm{adv}}}_{\text{Adversary}} + \underbrace{\varphi(\gamma_0 T_s)\,L_{\mathrm{hold}}\,T_s}_{\text{Sampled-Data Hold Rate}}.
\end{aligned}
\end{equation}

\paragraph{Tightened barrier, shell partition and certificate.}
With the decay rate \(\gamma_0\) of \eqref{eq:exp_weights}, the level-set-tightened reduced barrier, the tightened target set, and the un-dipped reduced safe set are
\begin{equation}
\label{eq:h_tilde_def}
\tilde{h}_r(z) := h(\Phi z) - \epsP^2 - \Delta h_{\mathrm{dip}}, \qquad
\tilde{\mathcal{C}}_r := \{z \in \ZO : \tilde{h}_r(z) \ge 0\}, \qquad
\mathcal{C}_r := \{z \in \ZO : h(\Phi z) \ge \epsP^2\},
\end{equation}
so \(\tilde{\mathcal{C}}_r \subseteq \mathcal{C}_r \subseteq \Crz\), and \(\mathcal{C}_r \subset \mathrm{int}(\Crz) = \{z^TPz < E_{\mathrm{safe}}\}\) because \(\epsP > 0\). The tail-tolerance hypothesis is
\begin{equation}
\label{eq:eps_basis_def}
\epsP^2 + \Delta h_{\mathrm{dip}} + L_h M_z T_s < E_{\mathrm{safe}} = \sup_{z \in \ZO} h(\Phi z),
\end{equation}
which is exactly the condition for the activation-threshold interval \(h_{\mathrm{act}} \in (L_hM_zT_s,\ E_{\mathrm{safe}} - \epsP^2 - \Delta h_{\mathrm{dip}})\) to be non-empty. For such \(h_{\mathrm{act}}\), partition \(\mathcal{C}_r\) into the active boundary shell and the passive interior core
\begin{equation}
\label{eq:shell}
\Cact := \{z \in \mathcal{C}_r : \tilde{h}_r(z) \le h_{\mathrm{act}}\}, \qquad
\Bint := \{z \in \mathcal{C}_r : \tilde{h}_r(z) > h_{\mathrm{act}}\} = \{z : \|z\|_P < r_{\min}\},
\end{equation}
where \(r_{\min}^2 := E_{\mathrm{safe}} - \epsP^2 - \Delta h_{\mathrm{dip}} - h_{\mathrm{act}} > 0\), so that \(\|z\|_P \ge r_{\min}\) on \(\Cact\). Because \(h_{\mathrm{act}} > L_hM_zT_s\), no trajectory segment that starts at a sample point \(z_k \in \Bint\) can reach \(\{\tilde h_r < 0\}\) within one period (Step~2 of Appendix~D). The Teacher is certified over a partition of \(\ZO\) (Proposition~\ref{prop:twin_crown}(ii)), and its certified CBF margin is \emph{defined} as
\begin{equation}
\label{eq:teacher_cbf}
\delta_{\mathrm{cert}} := \inf_{z \in \Cact} \Big[ \mathcal{L}_{f_r}\tilde{h}_r(z) + \mathcal{L}_{g_r}\tilde{h}_r(z)\Tt(z) + \gamma_0\,\tilde{h}_r(z) \Big].
\end{equation}
Because \(\Tt(z) \in U_{\mathrm{adm}}\), automatically \(\delta_{\mathrm{cert}} \le \delta_{\max} := \inf_{z \in \Cact}\sup_{u \in U_{\mathrm{adm}}}[\mathcal{L}_{f_r}\tilde{h}_r(z) + \mathcal{L}_{g_r}\tilde{h}_r(z)u + \gamma_0\tilde{h}_r(z)]\). On the port null space the control term vanishes (\(\mathcal{L}_{g_r}\tilde h_r(z) = -2(g_r^TPz)^T = 0\)), so \(\delta_{\mathrm{cert}} \ge \delta\) entails the necessary null-space condition
\begin{equation}
\label{eq:null_space_cond}
z^TQz - 2z^TPF_{\mathrm{nl},r}(z) + \gamma_0\tilde{h}_r(z) \ge \delta \qquad \forall z \in \Cact \text{ with } g_r^TPz = 0, \qquad Q := -(PA_{r,0} + A_{r,0}^TP),
\end{equation}
i.e., passive dissipation (\(z^TQz = \gamma\|z\|_P^2\) in class (H) with \(R_0 = \mathrm{blkdiag}(0,\gamma I_{K_m})\) at \(\epsilon = \gamma/2\), by Proposition~\ref{prop:dahlquist}) must carry the margin there; with \(P = P_0\) in class (H), \(z^TQ_0z = 2\gamma\|p_r\|_2^2\) vanishes on \(\{p_r = 0\}\), which is why \(P_0\) is generally not certifiable under momentum actuation and \(P_\epsilon\) is used.

\begin{lemma}[Closed-form certificate bound for full-rank modal actuation]
\label{lem:delta_cert_closed}
Let (H0), (H-sat), \(P = P_0 = I_K\), \(\Phi^T\Phi = I_K\), \(\Pf\Phi = \Phi\), \(g_r = I_K\) (\(m = K\)) hold, and \(c := 2(\lambda_{\min}(R_0) - L_{F_{\mathrm{nl}}} + \eta_{\mathcal{T}})\). Then
\begin{equation}
\label{eq:delta_cert_closed}
\delta_{\mathrm{cert}} \ge \min\Big\{c\,r_{\min}^2 + \gamma_0h_{\mathrm{act}},\ \ c\big(r_{\min}^2 + h_{\mathrm{act}} + \Delta h_{\mathrm{dip}}\big) - \gamma_0\Delta h_{\mathrm{dip}}\Big\},
\end{equation}
and the minimum equals the first branch if and only if \(c \ge \gamma_0\).
\end{lemma}
\begin{proof}
For \(z \in \Cact\), \(\mathcal{L}_{f_r}\tilde h_r(z) = -2z^T(J_0 - R_0)z - 2z^TF_{\mathrm{nl},r}(z) = 2z^TR_0z - 2z^T\Phi^TF_{\mathrm{nl}}(\Phi z)\) (skew term vanishes). Since \(0, \Phi z \in \mathcal{C} \subset \XO\) (Lemmas~\ref{lem:quad_split} and~\ref{lem:odd_containment}) and \(F_{\mathrm{nl}}(0) = 0\), Cauchy--Schwarz and Assumption~\ref{assump:generic} give \(|2z^T\Phi^TF_{\mathrm{nl}}(\Phi z)| \le 2\|z\|_2L_{F_{\mathrm{nl}}}\|\Phi z\|_2 = 2L_{F_{\mathrm{nl}}}\|z\|_2^2\); Rayleigh gives \(2z^TR_0z \ge 2\lambda_{\min}(R_0)\|z\|_2^2\); Lemma~\ref{lem:sector} with \(v = z\) gives \(-2z^T\Tt(z) \ge 2\eta_{\mathcal{T}}\|z\|_2^2\). Hence the bracket in \eqref{eq:teacher_cbf} is at least \(c\rho + \gamma_0\tilde h_r(z)\) with \(\rho := \|z\|_2^2\). On \(\Cact\), \(\tilde h_r(z) = r_{\min}^2 + h_{\mathrm{act}} - \rho\) and \(\rho \in [r_{\min}^2, r_{\min}^2 + h_{\mathrm{act}} + \Delta h_{\mathrm{dip}}]\), so the lower bound \((c - \gamma_0)\rho + \gamma_0(r_{\min}^2 + h_{\mathrm{act}})\) is affine in \(\rho\) and its infimum is attained at an endpoint: \(\rho = r_{\min}^2\) gives the first branch, \(\rho = r_{\min}^2 + h_{\mathrm{act}} + \Delta h_{\mathrm{dip}}\) the second. Their difference (first minus second) is \(-(c - \gamma_0)(h_{\mathrm{act}} + \Delta h_{\mathrm{dip}})\), which is \(\le 0\) iff \(c \ge \gamma_0\).
\end{proof}

\subsection{Deadzone Practical Stability and Full-State Sampled-Data CBF Invariance}

Let \(\overline{\varepsilon}_q := \max\big(\eA,\ \eC + \dul\big) + \dul\), and, when \(F_{\mathrm{nl}}\) is \(C^1\) on a neighbourhood of \(\Phi\Crz\), let \(\Lsec := \sup_{z \in \Crz}\|P^{1/2}\nabla F_{\mathrm{nl},r}(z)P^{-1/2}\|_2\) be the \(P\)-norm sector modulus.

\begin{proposition}[Deadzone-Sector Limit-Cycle Elimination]
\label{prop:deadzone_stability}
Let \(P \in \{P_0, P_\epsilon\}\) and \(\mu_R > 0\) be as in Proposition~\ref{prop:dahlquist} (so \(\mu_{2,P}(A_{r,0}) \le -\mu_R\)), let \(\Tt\) be the Teacher \eqref{eq:teacher_def} under (H-sat) (so \(\Tt(0) = 0\) and \(z^TPg_r\Tt(z) \le 0\) on \(\ZO\) by Lemma~\ref{lem:sector}), let \(F_{\mathrm{nl}}\) be \(C^1\) near \(\Phi\Crz\), and assume
(i) \(\mu_R > \Lsec\); (ii) \(\|u(t) - \Tt(z(t))\|_2 \le \overline{\varepsilon}_q\) for all \(t\) in an interval \(I\) on which a solution of the reduced closed loop with \(w \equiv 0\), \(x_\perp \equiv 0\) satisfies \(z(t) \in \Crz\). Then, with
\begin{equation}
\label{eq:rss_ball}
\mathcal{B}_{\mathrm{ss}}(r_{\mathrm{ss}}) := \left\{ z \in \mathbb{R}^K : \|z\|_P \le r_{\mathrm{ss}} := \frac{\sqrt{\lambda_{\max}(P)}\,\|g_r\|_2\,\overline{\varepsilon}_q}{\mu_R - \Lsec} \right\},
\end{equation}
(a) \(D^+\|z(t)\|_P \le -(\mu_R - \Lsec)\|z(t)\|_P + \sqrt{\lambda_{\max}(P)}\|g_r\|_2\overline{\varepsilon}_q\) for all \(t \in I\) (upper right Dini derivative; at \(z = 0\) as well); (b) \(\|z(t)\|_P \le \max(\|z(t_0)\|_P, r_{\mathrm{ss}})\) for \(t \ge t_0\) in \(I\), so \(\mathcal{B}_{\mathrm{ss}}(r_{\mathrm{ss}})\cap\Crz\) is forward invariant, and \(\|z(t)\|_P > r_{\mathrm{ss}}\) forces strict decrease; (c) no periodic orbit (in particular no quantization-induced limit cycle) contained in \(\Crz\) meets \(\Crz \setminus \mathcal{B}_{\mathrm{ss}}(r_{\mathrm{ss}})\). Under the hypotheses of Theorem~\ref{thm:main_cbf}, (ii) holds along the closed loop. If moreover
\begin{equation}
\label{eq:deadzone_cond}
\text{(H-dz)}\qquad r_{\mathrm{ss}} < r_{\min},
\end{equation}
then \(\mathcal{B}_{\mathrm{ss}}(r_{\mathrm{ss}}) \subset \Bint\).
\end{proposition}

\begin{theorem}[Full-State Continuous-Time Safety Under Multi-Rate Quantized Control]
\label{thm:main_cbf}
Assume, with every constant evaluated as defined above,
\begin{enumerate}
\item[(H0)] ODD containment \eqref{eq:H0} and \(E_{\mathrm{safe}} < \frac14\DO^2\lambda_{\min}(P)\);
\item[(H1)] Assumption~\ref{assump:generic}, with \(L_{F_{\mathrm{nl}}}\), \(M_\perp\), \(\Lperp\) taken over \(\XO\);
\item[(H2)] Assumption~\ref{assump:tail_iss} ((H2a)--(H2c)); in class (H), \(\gamma < 2\omega_{\min}\) and \(\epsilon = \gamma/2\);
\item[(H3)] the Teacher construction \eqref{eq:teacher_weights}--\eqref{eq:teacher_def} with \(H/2 \ge m\), \(G_{\mathrm{port}} = g_r^TP^2g_r \succ 0\), and (H-sat) \eqref{eq:sat_cond};
\item[(H4)] the tail tolerance \eqref{eq:eps_basis_def} and \(h_{\mathrm{act}} \in (L_hM_zT_s,\ E_{\mathrm{safe}} - \epsP^2 - \Delta h_{\mathrm{dip}})\) (equivalently \(r_{\min}^2 > 0\));
\item[(H5)] (H-dz) \eqref{eq:deadzone_cond};
\item[(H6)] \(b_A \ge 1\) satisfies \eqref{eq:lut_capacity} (the lower bound of \eqref{eq:codesign_window} is optional);
\item[(H7)] the Certified Control Authority Condition
\begin{equation}
\label{eq:control_authority}
\delta_{\mathrm{total}}(K, b_A, b_B, T_s) \le \delta_{\mathrm{cert}} \;\big(\le \delta_{\max}\big),
\end{equation}
with \(\delta_{\mathrm{cert}}\) defined by \eqref{eq:teacher_cbf};
\item[(H8)] initial conditions \(x(0) = \Phi z(0) + x_\perp(0)\) with \(z(0) \in \tilde{\mathcal{C}}_r\) and \(\|x_\perp(0)\|_{\Pfe} \le M_\perp/\lambda_{K+1}\).
\end{enumerate}
Then the Carath\'eodory solution \(x(t)\) of \eqref{eq:full_system} (\(\Sigma \equiv 0\)) under the multi-rate control law \eqref{eq:multirate_hold} and any measurable \(w(t) \in \mathcal{B}(\varepsilon_{\mathrm{adv}})\) exists on \([0,\infty)\) and satisfies, for all \(t \ge 0\) and \(k \in \mathbb{N}_0\): (a) \(\|x_\perp(t)\|_{\Pfe} \le \epsP\); (b) \(\tilde{h}_r(z(t_k)) \ge 0\) and \(\tilde h_r(z(t)) \ge -\Delta h_{\mathrm{dip}}\), i.e.\ \(z(t) \in \mathcal{C}_r\); (c) \(h(x(t)) \ge \tilde h_r(z(t)) + \Delta h_{\mathrm{dip}} \ge 0\), i.e.\ \(x(t) \in \mathcal{C}\). Hypothesis (H5) is not used for (a)--(c); it makes the deadzone analysis of Proposition~\ref{prop:deadzone_stability} consistent with the shell partition for the same design.
\end{theorem}

\begin{remark}[Spectral cutoff for RDA systems]
\label{rem:rda_cutoff}
This remark is not part of Theorem~\ref{thm:main_cbf}. For the RDA class of Remark~\ref{rem:tail_rates}(ii) let \(\mathrm{Pe} := c_{\mathrm{adv}}\ell/\nu\), a tail tolerance \(\bar\varepsilon > 0\), and a coupling budget \(\bar\delta_\perp > 0\). Define \(K_{\min}(\mathrm{Pe})\) as the smallest \(K\) with both \(\epsP(K) \le \bar\varepsilon\) and \(\Lperp(K)\,\epsP(K) \le \bar\delta_\perp\). With the continuum rate, \(\epsP(K) = M_\perp(K)/(\alpha_0 + \nu(K+1)^2\pi^2/\ell^2)\), so the tail-tolerance component is the exact inversion
\(\alpha_0 + \nu\pi^2(K+1)^2/\ell^2 \ge M_\perp(K)/\bar\varepsilon\). \emph{\(H_0^1\)-bounded sets} (\(M_\perp\) independent of \(K\)): \((K+1) \ge \frac{\ell}{\pi}\sqrt{(M_\perp/\bar\varepsilon - \alpha_0)_+/\nu}\), which is \(\frac{\ell}{\pi}\sqrt{M_\perp/(\nu\bar\varepsilon)}\) for \(\alpha_0 = 0\) and scales as \(O(\sqrt{\mathrm{Pe}})\) when \(M_\perp \propto c_{\mathrm{adv}}\). \emph{\(L^2\)-bounded sets} (\(M_\perp = c_0c_{\mathrm{adv}}(K+1)\)): the inequality is quadratic in \(K+1\) with no square-root law; for \(\alpha_0 = 0\) it reads \(K + 1 \ge c_0\mathrm{Pe}\,\ell/(\pi^2\bar\varepsilon)\) (\(= c_0\mathrm{Pe}/(\pi^2\bar\varepsilon)\) for unit length), and for \(\alpha_0 \neq 0\), \(K+1\) must exceed the larger root \(\frac{\ell^2}{2\pi^2\nu}\big[\frac{c_0c_{\mathrm{adv}}}{\bar\varepsilon} + \big((\frac{c_0c_{\mathrm{adv}}}{\bar\varepsilon})^2 - \frac{4\alpha_0\nu\pi^2}{\ell^2}\big)_+^{1/2}\big]\). \emph{Coupling component:} integrating by parts against the sine modes, \(\|\Phi^T\partial_\xi x_\perp\|_2 \le \frac{K\pi}{\ell}\|x_\perp\|_2\), so \(\Lperp(K) \le c_{\mathrm{adv}}K\pi/\ell + L_{F_{\mathrm{nl}}}\); hence on \(H_0^1\)-bounded sets \(\Lperp\epsP \lesssim c_{\mathrm{adv}}M_\perp\ell/(\pi\nu K) \propto \mathrm{Pe}/K\), requiring \(K = O(\mathrm{Pe})\), while on \(L^2\)-bounded sets the product saturates at a \(K\)-independent level \(\propto c_{\mathrm{adv}}\mathrm{Pe}\), so the coupling budget is a condition on \((\mathrm{Pe}, \bar\delta_\perp)\) that increasing \(K\) cannot repair. The overall cutoff scaling is therefore \(O(\mathrm{Pe})\) (when achievable), and \(O(\sqrt{\mathrm{Pe}})\) describes only the tail-tolerance component. Since advection is skew in \(L^2\) but not in \(H_0^1\), and \(x^T\Pf J_{\mathrm{full}}\Pf x \equiv 0\), the advective cross-coupling cancels exactly in the full-energy derivative; a full-state barrier could exploit this (future work).
\end{remark}

\begin{remark}[LUT indexing variants and measurement rounding]
\label{rem:lut_variants}
(i) \emph{Port-space LUT} (\(m < K\)): since \(\Tt\) depends on \(z\) only through \(v = g_r^TPz\), Path A may index the table by \(Q_{b_A}(v)\). Replace \(K \to m\), \(\DO \to D_v := 2\sup_{z\in\Crz}\|g_r^TPz\|_\infty\), \(M_z \to M_v := \|g_r^TP\|_2M_z\) and \(\LTz \to L_{\mathcal{T}}\) in \eqref{eq:lut_capacity}, \eqref{eq:eps_A_triangle} and \eqref{eq:codesign_window}; the window then guarantees \(L_{\mathcal{T}}\sqrt{m}D_v/2^{b_A+1} < L_{\mathcal{T}}M_vT_s = \dul\). (ii) \emph{Hashed boundary-shell indexing:} for a fixed \(b_A\) (after \(\Delta h_{\mathrm{dip}}\) and \(\Cact\) are fixed, so no definitional cycle arises), let \(N_{\mathrm{shell}}\) be the number of lattice cells meeting \(\Cact\); store only these under a collision-free (perfect) hash built offline, with \(\lceil\log_2N_{\mathrm{shell}}\rceil \le B_{\max}\) address bits plus key-check bits per entry so that misses occur only for queries outside the shell. On a miss (in particular for \(z_k \in \Bint\)) Path A does not fire and the register keeps \(u_{B,k-1}\) on \([t_k + \tau_A, t_k + \tau_B)\), with error \(\le \eC + \dul\); this is harmless because the certificate is not invoked when \(z_k \in \Bint\) (Step~2, Appendix~D), and \(\overline{\varepsilon}_q\) already covers it. (iii) \emph{Measurement rounding:} Theorem~\ref{thm:main_cbf} assumes exact acquisition of \(z_k\); input rounding for Path B is modelled by \(e_0\) in Proposition~\ref{prop:twin_crown}. If the Path-A index is computed from \(\hat z_k = z_k + e_z\) with \(\|e_z\|_2 \le \varepsilon_z\), replace \(\eA\) by \(\eA + \varepsilon_{\mathrm{meas}}\), \(\varepsilon_{\mathrm{meas}} := \LTz\varepsilon_z\).
\end{remark}

\begin{remark}[Isomorphism to Asymmetric Zero-Sum HJI Differential Safety Games]
\label{rem:hji_game}
Because \eqref{eq:full_system} and \eqref{eq:reduced_dynamics} are separately affine in \(u \in U_{\mathrm{adm}}\) and \(w \in \mathcal{B}(\varepsilon_{\mathrm{adv}})\), the continuous-time Hamilton--Jacobi--Isaacs (HJI) safety Hamiltonian satisfies Isaacs' minimax condition identically (\(\sup_{u \in U_{\mathrm{adm}}} \inf_{w \in \mathcal{B}(\varepsilon_{\mathrm{adv}})} \dot{\tilde{h}}_r = \inf_{w \in \mathcal{B}(\varepsilon_{\mathrm{adv}})} \sup_{u \in U_{\mathrm{adm}}} \dot{\tilde{h}}_r\)). Consequently, Theorem~\ref{thm:main_cbf} certifies winning safety invariance for an asymmetric two-player zero-sum differential game in which a continuous-time, full-information adversary applies the pointwise worst-case disturbance \(w^*(x(t)) = -\varepsilon_{\mathrm{adv}}\,\mathcal{L}_{d_r}\tilde{h}_r(z(t))^T / \|\mathcal{L}_{d_r}\tilde{h}_r(z(t))\|_2\) when \(\mathcal{L}_{d_r}\tilde{h}_r(z(t)) \neq 0\) (and \(w^*(x(t)) = 0\) otherwise, plus worst-case tail excitation within \(M_\perp\)) inside the inter-sample window \([t_k, t_{k+1})\) against the quantized, multi-rate sampled-data defender \eqref{eq:multirate_hold}. The feedback \(w^*\) is discontinuous, so closed-loop solutions are understood in the Filippov sense; since the Filippov regularization takes convex combinations of values in the convex ball \(\mathcal{B}(\varepsilon_{\mathrm{adv}})\), every Filippov solution is a Carath\'eodory solution for some measurable \(w(t) \in \mathcal{B}(\varepsilon_{\mathrm{adv}})\), which is all the proof uses. Under the distributional drift of Corollary~\ref{cor:stochastic_w2} the ball is \(\mathcal{B}(\varepsilon_{\mathrm{adv}}(T))\).
\end{remark}

\begin{corollary}[Stochastic It\^o Sampled-Data Safety and Wasserstein-2 Horizon]
\label{cor:stochastic_w2}
Consider the stochastic system \eqref{eq:full_system} over a finite horizon \(T = N_T T_s > 0\) with \(\Sigma\) globally Lipschitz (so strong solutions exist and are unique under the adapted control \eqref{eq:multirate_hold}), projected diffusion \(\Sigma_r(x) := P_0^{-1}\Phi^T\Pf\Sigma(x)\) and tail diffusion \(\Sigma_\perp(x) := P_\perp\Sigma(x)\), bounded over the safe set by \(\overline{\sigma}_r := \sup_{x \in \mathcal{C}} \|\Sigma_r(x)\|_2 < \infty\) and by the trace-class (Hilbert--Schmidt in the \(\Pfe\)-metric) bound \(\overline{\sigma}_\perp^2 := \sup_{x \in \mathcal{C}}\|\Pfe^{1/2} P_\perp \Sigma(x)\|_F^2 < \infty\).\footnote{The bound is valid but \(N\)-dependent: for spatially white noise \(\|\Pfe^{1/2}P_\perp\Sigma\|_F = \sigma\sqrt{N-K}\) when \(\Pfe = I\); mode-weighted functionals give \(\sum_k\sigma_k^2/(2\lambda_k)\) instead. Space--time white noise in one dimension violates the hypothesis in the continuum limit.} The exogenous disturbance decomposes as \(w(t) = w_{\mathrm{adv}}(t) + \int_\Xi \psi(\xi)\,d\mu_t(\xi)\), where \(w_{\mathrm{adv}}\) is progressively measurable with \(w_{\mathrm{adv}}(t) \in \mathcal{B}(\varepsilon_{\mathrm{adv}})\), \((\Xi, d_\Xi)\) is a Polish space, \(\psi : \Xi \to \mathbb{R}^{n_w}\) is \(1\)-Lipschitz from \(d_\Xi\) to \(\ell_2\) with \(\int_\Xi \psi\,d\mu_0 = 0\), and the laws \(\mu_t\) have finite second moments and drift at Wasserstein-2 rate \(W_2(\mu_t, \mu_0) \le r_{W_2} t\). Set \(\varepsilon_{\mathrm{adv}}(T) := \varepsilon_{\mathrm{adv}} + r_{W_2}T\) and the drift-enlarged constants
\[
M_\perp(T) := M_\perp + \|\Pfe^{1/2}P_\perp D\|_2\,r_{W_2}T, \qquad \epsP(T) := M_\perp(T)/\lambda_{K+1}, \qquad \eta := \frac{\lambda_{K+1}}{2\overline{\sigma}_\perp^2},
\]
the continuous-time \((1 - \alpha/4)\) Ornstein--Uhlenbeck tail envelope (in the \(\Pfe\)-norm)
\[
\varepsilon_{\mathrm{basis},\sigma}(K, \alpha, T) := \epsP(T) + \Delta_\sigma, \qquad \big(1 + \tfrac12\lambda_{K+1}T\big)e^{-\eta\Delta_\sigma^2} = \tfrac{\alpha}{4}, \ \text{ i.e. }\ \Delta_\sigma = \overline{\sigma}_\perp\sqrt{\frac{2\ln\!\big(4(1 + \frac{1}{2}\lambda_{K+1}T)/\alpha\big)}{\lambda_{K+1}}},
\]
and, globally replacing \(\epsP\) by \(\epsS\) and \(\varepsilon_{\mathrm{adv}}\) by \(\varepsilon_{\mathrm{adv}}(T)\) in \(M_z\) (so \(M_z(T) \le M_z + \|d_r\|_2r_{W_2}T + \Lperp(\epsS - \epsP)\)), \(L_{\mathrm{hold}}\), \(\dul\), the window \eqref{eq:codesign_window}, \eqref{eq:delta_dip}, \eqref{eq:h_tilde_def}, \eqref{eq:shell}, \(r_{\min}\) and \eqref{eq:teacher_cbf}, but keeping \(\overline{d}_h\varepsilon_{\mathrm{adv}}\) in the adversary term, denote the resulting margin \eqref{eq:delta_total_exact} by \(\delta_{\mathrm{total},\sigma}\) (it reduces to \(\delta_{\mathrm{total}}\) when \(\overline{\sigma}_\perp = 0\), \(r_{W_2} = 0\)). Let \(T_\perp := \sup\{T' > 0 : \epsS(T')^2 + \Delta h_{\mathrm{dip}}(T') + L_hM_z(T')T_s < E_{\mathrm{safe}}\}\). Suppose \(h\) is as in \eqref{eq:h_def} (so \(h \in C^\infty\)), \(\mathrm{tr}\big(\Sigma_r(x)^T \nabla_z^2 \tilde{h}_r(z) \Sigma_r(x)\big) \ge -2\,\delta_{\mathrm{It\hat{o}}}\) and \(\sigma_h := \sup_{x \in \mathcal{C}} \|\nabla_z\tilde{h}_r(z)^T\Sigma_r(x)\|_2 < \infty\), \(\overline{h}_r := h_{\mathrm{act}}\) with \(h_{\mathrm{act}} > L_h(M_z(T) T_s + r_{\mathrm{BM}}(\alpha))\), \(\|x_\perp(0)\|_{\Pfe} \le M_\perp(T)/\lambda_{K+1}\), and \(z(0) \in \mathrm{int}(\tilde{\mathcal{C}}_r)\) (non-empty) is deterministic. Define the inter-sample Brownian excursion radius \(r_{\mathrm{BM}}(\alpha) := \overline{\sigma}_r \sqrt{2 K T_s \ln(8 K N_T / \alpha)}\), the pointwise multi-rate excess \(\Delta\delta_{\mathrm{pw}} := \overline{g}_h\max(\varepsilon_A(b_A), \varepsilon_{\mathrm{CROWN}}(b_B) + \Delta u_{\mathrm{lag}}) - \overline{g}_h[w_2\varepsilon_A(b_A) + (1-w_2)\varepsilon_{\mathrm{CROWN}}(b_B) + w_1\Delta u_{\mathrm{lag}}] + (1 - \varphi(\gamma_0 T_s))L_{\mathrm{hold}}T_s + (L_{\mathcal{L}_{f_r}\tilde{h}_r} + L_{\mathcal{L}_{g_r}\tilde{h}_r}u_{\max} + \gamma_0 L_h + \overline{g}_h \LTz)r_{\mathrm{BM}}(\alpha)\), the drift process \(\mu_B(t) := \nabla\tilde h_r(z_t)^T[f_r(z_t) + g_ru(t) + d_rw(t) + \Delta_\perp(z_t, x_{\perp,t})]\), the reduced activation level \(h_{\mathrm{act}}^- := h_{\mathrm{act}} - L_h(M_z(T)T_s + r_{\mathrm{BM}}(\alpha)) > 0\), the reference level \(h_0 \in (0, h_{\mathrm{act}}]\), the Kushner exponent \(\thK := \frac{\ln(2/\alpha)}{h_0}\), the worst-case negative interior drift
\[
\begin{aligned}
\underline{\mu}^-_{\mathrm{int}} := \sup\Big\{&\big(-\nabla\tilde h_r(z)^T[f_r(z) + g_ru + d_rw + \Delta_\perp(z,x_\perp)]\big)_+ :\ z \in \Crz,\ \tilde h_r(z) > h_{\mathrm{act}}^-,\ u \in U_{\mathrm{adm}},\\
&\|w\|_2 \le \varepsilon_{\mathrm{adv}}(T),\ x_\perp \in \mathcal{V}_\perp,\ \|x_\perp\|_{\Pfe} \le \epsS,\ \Phi z + x_\perp \in \mathcal{C}\Big\},
\end{aligned}
\]
the passive interior generator bound \(c_{\mathrm{int}} := \thK\big(\underline{\mu}^-_{\mathrm{int}} + \delta_{\mathrm{It\hat{o}}} + \frac{1}{2}\thK\sigma_h^2\big)e^{-\thK(h_{\mathrm{act}}^- + \Delta h_{\mathrm{dip}})}\), and
\begin{equation}
\label{eq:delta_stoch}
\delta_{\mathrm{stoch}}(\alpha) := \Delta\delta_{\mathrm{pw}} + \gamma_0 \overline{h}_r + \delta_{\mathrm{It\hat{o}}} + \frac{\ln(2/\alpha)}{2\,h_0}\,\sigma_h^2.
\end{equation}
If, in addition to the hypotheses of Theorem~\ref{thm:main_cbf} with the above replacements, \(\delta_{\mathrm{total},\sigma} + \delta_{\mathrm{stoch}}(\alpha) + \delta_{\mathrm{drift}} \le \delta_{\mathrm{cert}}\) with drift reserve \(\delta_{\mathrm{drift}} > 0\), the initial level \(\beta_0 := \tilde{h}_r(z(0)) + \Delta h_{\mathrm{dip}}\) satisfies \(\beta_0 > h_0\), and
\begin{equation}
\label{eq:T_TTL}
T \le T_{\mathrm{TTL}} := \min\Big\{\frac{\delta_{\mathrm{drift}}}{\overline{d}_h\,r_{W_2}},\ \ \frac{\alpha\big(1 - e^{-\thK(\beta_0 - h_0)}\big)}{2c_{\mathrm{int}}},\ \ T_\perp\Big\}
\end{equation}
(terms with zero denominator read \(+\infty\); the second bound is equivalent to \(\beta_0 \ge h_0 - \thK^{-1}\ln(1 - 2c_{\mathrm{int}}T/\alpha)\) with a positive logarithm argument), then the full-order trajectory satisfies \(\mathbb{P}\big(\inf_{t \in [0, T]} h(x(t)) \ge 0\big) \ge 1 - \alpha\).
\end{corollary}
